Database

Ignite provides a powerful database layer with both a fluent query builder and an Eloquent-style ORM. Supports MySQL, PostgreSQL, SQLite, and SQL Server.

Configuration

Configure your database connection in .env:

# SQLite (default)
DB_CONNECTION=sqlite
DB_DATABASE=database.db

# MySQL
DB_CONNECTION=mysql
DB_HOST=127.0.0.1
DB_PORT=3306
DB_DATABASE=ignite
DB_USERNAME=root
DB_PASSWORD=secret

# PostgreSQL
DB_CONNECTION=postgres
DB_HOST=127.0.0.1
DB_PORT=5432
DB_DATABASE=ignite
DB_USERNAME=postgres
DB_PASSWORD=secret
DB_SSLMODE=disable

Query Builder

The query builder provides a fluent interface for building SQL queries:

Retrieving Data

import "github.com/sazzad/ignite/database"

// Get all records
users := database.Table("users").Get()

// Get first record
user := database.Table("users").First()

// Get specific columns
users := database.
    Table("users").
    Select("id", "name", "email").
    Get()

// Get single value
email := database.
    Table("users").
    Where("id", 1).
    Value("email")

// Pluck a column
emails := database.
    Table("users").
    Pluck("email")

Where Clauses

// Simple where
users := database.
    Table("users").
    Where("status", "active").
    Get()

// Where with operator
users := database.
    Table("users").
    Where("age", ">=", 18).
    Get()

// Multiple conditions (AND)
users := database.
    Table("users").
    Where("status", "active").
    Where("role", "admin").
    Get()

// OR conditions
users := database.
    Table("users").
    Where("role", "admin").
    OrWhere("role", "moderator").
    Get()

// Where In
users := database.
    Table("users").
    WhereIn("id", []int{1, 2, 3}).
    Get()

// Where Between
users := database.
    Table("users").
    WhereBetween("age", 18, 65).
    Get()

// Where Null
users := database.
    Table("users").
    WhereNull("deleted_at").
    Get()

Joins

// Inner join
users := database.
    Table("users").
    Join("profiles", "users.id", "=", "profiles.user_id").
    Select("users.*", "profiles.bio").
    Get()

// Left join
users := database.
    Table("users").
    LeftJoin("orders", "users.id", "=", "orders.user_id").
    Get()

// Complex join conditions
users := database.
    Table("users").
    Join("contacts", func(j *database.JoinClause) {
        j.On("users.id", "=", "contacts.user_id")
        j.Where("contacts.active", true)
    }).
    Get()

Ordering & Grouping

// Order by
users := database.
    Table("users").
    OrderBy("created_at", "desc").
    Get()

// Latest / Oldest
users := database.Table("users").Latest().Get()
users := database.Table("users").Oldest().Get()

// Group by
stats := database.
    Table("orders").
    Select("user_id", database.Raw("COUNT(*) as total")).
    GroupBy("user_id").
    Get()

// Having
users := database.
    Table("orders").
    GroupBy("user_id").
    Having("COUNT(*)", ">", 5).
    Get()

Limiting & Pagination

// Limit
users := database.Table("users").Limit(10).Get()

// Offset
users := database.
    Table("users").
    Offset(20).
    Limit(10).
    Get()

// Paginate
paginator := database.
    Table("users").
    Paginate(15) // 15 per page

Aggregates

// Count
count := database.Table("users").Count()

// Max, Min, Sum, Avg
max := database.Table("orders").Max("amount")
min := database.Table("orders").Min("amount")
sum := database.Table("orders").Sum("amount")
avg := database.Table("orders").Avg("amount")

// Exists
exists := database.
    Table("users").
    Where("email", "john@example.com").
    Exists()

Inserts

// Insert single record
id := database.
    Table("users").
    Insert(map[string]interface{}{
        "name":  "John Doe",
        "email": "john@example.com",
    })

// Insert multiple records
database.Table("users").Insert([]map[string]interface{}{
    {"name": "John", "email": "john@example.com"},
    {"name": "Jane", "email": "jane@example.com"},
})

Updates

// Update records
database.
    Table("users").
    Where("id", 1).
    Update(map[string]interface{}{
        "name": "John Updated",
    })

// Increment / Decrement
database.Table("posts").Where("id", 1).Increment("views")
database.Table("products").Where("id", 1).Decrement("stock", 5)

Deletes

// Delete records
database.
    Table("users").
    Where("id", 1).
    Delete()

// Truncate table
database.Table("sessions").Truncate()

ORM (Eloquent)

Ignite's ORM provides an elegant ActiveRecord implementation for working with your database:

Defining Models

package models

import (
    "github.com/sazzad/ignite/orm"
    "time"
)

type User struct {
    orm.Model
    Name      string
    Email     string
    Password  string
    Age       int
    Active    bool
    Posts     []Post
    Profile   *Profile
    CreatedAt time.Time
    UpdatedAt time.Time
}

// Specify table name (optional)
func (User) TableName() string {
    return "users"
}

Retrieving Models

// Find by primary key
var user User
database.Find(&user, 1)

// Find with conditions
var user User
database.Where("email = ?", "john@example.com").First(&user)

// Get all records
var users []User
database.Find(&users)

// With query builder methods
var users []User
database.
    Where("active = ?", true).
    Order("created_at desc").
    Limit(10).
    Find(&users)

Creating Models

user := User{
    Name:  "John Doe",
    Email: "john@example.com",
    Age:   30,
}
database.Create(&user)

// user.ID is now populated

Updating Models

var user User
database.Find(&user, 1)

// Update single field
database.Model(&user).Update("name", "John Updated")

// Update multiple fields
database.Model(&user).Updates(User{
    Name: "John Updated",
    Age:  31,
})

// Save (updates all fields)
user.Name = "John Updated"
database.Save(&user)

Deleting Models

var user User
database.Find(&user, 1)
database.Delete(&user)

// Soft deletes (if using soft delete trait)
database.Delete(&user) // Sets deleted_at

// Force delete (permanent)
database.Unscoped().Delete(&user)

Relationships

// One-to-One
type User struct {
    orm.Model
    Profile *Profile `gorm:"foreignKey:UserID"`
}

// One-to-Many
type User struct {
    orm.Model
    Posts []Post `gorm:"foreignKey:UserID"`
}

// Many-to-Many
type User struct {
    orm.Model
    Roles []Role `gorm:"many2many:user_roles;"`
}

// Eager loading
var user User
database.Preload("Posts").First(&user, 1)

// Nested eager loading
database.Preload("Posts.Comments").First(&user, 1)

Scopes

// Define scope
func ActiveUsers(db *orm.DB) *orm.DB {
    return db.Where("active = ?", true)
}

func RecentUsers(db *orm.DB) *orm.DB {
    return db.Where("created_at > ?", time.Now().AddDate(0, 0, -30))
}

// Use scopes
var users []User
database.
    Scopes(ActiveUsers, RecentUsers).
    Find(&users)

Migrations

Migrations provide version control for your database schema:

Creating Migrations

ignite make:migration create_users_table

Migration Structure

package migrations

import "github.com/sazzad/ignite/schema"

func CreateUsersTable() *schema.Migration {
    return &schema.Migration{
        Up: func(s *schema.Schema) {
            s.Create("users", func(t *schema.Blueprint) {
                t.ID()
                t.String("name")
                t.String("email").Unique()
                t.String("password")
                t.Integer("age").Nullable()
                t.Boolean("active").Default(true)
                t.Timestamps()
            })
        },
        Down: func(s *schema.Schema) {
            s.DropIfExists("users")
        },
    }
}

Available Column Types

t.ID()                           // Auto-incrementing primary key
t.String("name")                  // VARCHAR(255)
t.Text("description")           // TEXT
t.Integer("age")                 // INTEGER
t.BigInteger("views")            // BIGINT
t.Float("price")                 // FLOAT
t.Decimal("amount", 8, 2)         // DECIMAL(8,2)
t.Boolean("active")              // BOOLEAN
t.Date("birth_date")            // DATE
t.DateTime("published_at")       // DATETIME
t.Timestamp("created_at")        // TIMESTAMP
t.JSON("metadata")              // JSON
t.Timestamps()                    // created_at + updated_at
t.SoftDeletes()                   // deleted_at

Running Migrations

# Run all pending migrations
ignite migrate

# Rollback last migration
ignite migrate:rollback

# Rollback all migrations
ignite migrate:reset

# Rollback and re-run all migrations
ignite migrate:refresh

# Drop all tables and re-run migrations
ignite migrate:fresh

Transactions

Execute operations within a database transaction:

err := database.Transaction(func(tx *database.DB) error {
    // Create user
    user := User{Name: "John", Email: "john@example.com"}
    if err := tx.Create(&user).Error; err != nil {
        return err // Rollback
    }

    // Create profile
    profile := Profile{UserID: user.ID, Bio: "..."}
    if err := tx.Create(&profile).Error; err != nil {
        return err // Rollback
    }

    return nil // Commit
})

if err != nil {
    // Transaction failed and was rolled back
}
