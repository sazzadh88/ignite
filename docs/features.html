Features

Ignite includes a comprehensive suite of features to accelerate your development. Everything you need to build modern web applications.

Validation

Powerful validation with 50+ built-in rules and support for custom rules:

Basic Validation

import "github.com/sazzadh88/ignite/validation"

validator := validation.Make(req.All(), map[string]string{
    "name":  "required|min:3|max:255",
    "email": "required|email|unique:users,email",
    "age":   "required|integer|min:18",
    "url":   "required|url",
})

if validator.Fails() {
    return req.Back().WithErrors(validator.Errors())
}

Available Rules

required, required_if, required_unless
min, max, between, size
email, url, ip, alpha, alpha_dash, alpha_num
numeric, integer, digits, digits_between
in, not_in
unique, exists
confirmed (password_confirmation)
date, date_format, before, after
regex, not_regex
array, json
accepted, boolean
same, different
file, image, mimes, max (file size)

Form Request Validation

package requests

type CreateUserRequest struct {}

func (r *CreateUserRequest) Rules() map[string]string {
    return map[string]string{
        "name":  "required|min:3",
        "email": "required|email|unique:users",
    }
}

func (r *CreateUserRequest) Messages() map[string]string {
    return map[string]string{
        "name.required": "Please provide a name",
    }
}

// Use in controller
func CreateUser(req *http.Request, form *CreateUserRequest) *http.Response {
    // Request is already validated
}

Cache

Unified caching interface with multiple drivers (memory, Redis, file):

Basic Usage

import "github.com/sazzadh88/ignite/cache"

// Store item
cache.Put("key", "value", 60) // 60 seconds

// Retrieve item
value := cache.Get("key")

// Get with default
value := cache.Get("key", "default")

// Check existence
if cache.Has("key") {
    // Key exists
}

// Delete item
cache.Forget("key")

// Clear all
cache.Flush()

Remember Pattern

// Fetch or store
users := cache.Remember("users", 300, func() interface{} {
    return database.Table("users").Get()
})

// Remember forever
config := cache.RememberForever("config", func() interface{} {
    return loadConfig()
})

Cache Tags

// Store with tags
cache.
    Tags("users", "admins").
    Put("user:1", user, 60)

// Retrieve with tags
user := cache.Tags("users").Get("user:1")

// Flush by tag
cache.Tags("users").Flush()

Atomic Locks

// Acquire lock
lock := cache.Lock("process-orders", 10)

if lock.Get() {
    defer lock.Release()

    // Do work...
}

Queue & Jobs

Process background jobs with multiple drivers (memory, Redis, database):

Creating Jobs

package jobs

import "github.com/sazzadh88/ignite/queue"

type ProcessPayment struct {
    OrderID int
    Amount  float64
}

func (j *ProcessPayment) Handle() error {
    // Process payment logic
    return nil
}

// Customize job
func (j *ProcessPayment) Queue() string {
    return "payments"
}

func (j *ProcessPayment) Retries() int {
    return 3
}

func (j *ProcessPayment) Timeout() int {
    return 30 // seconds
}

Dispatching Jobs

// Dispatch immediately
queue.Dispatch(&ProcessPayment{OrderID: 123, Amount: 99.99})

// Dispatch to specific queue
queue.OnQueue("payments").Dispatch(&ProcessPayment{})

// Delay dispatch
queue.Delay(60).Dispatch(&ProcessPayment{}) // 60 seconds

Job Chaining

// Chain jobs (run sequentially)
queue.
    Chain(
        &ProcessPayment{OrderID: 123},
        &SendReceipt{OrderID: 123},
        &UpdateInventory{OrderID: 123},
    ).
    Dispatch()

Job Batching

// Process jobs in batch
batch := queue.Batch(
    &SendEmail{UserID: 1},
    &SendEmail{UserID: 2},
    &SendEmail{UserID: 3},
).Then(func() {
    // All jobs completed
}).Catch(func(err error) {
    // One or more jobs failed
}).Dispatch()

Running Workers

# Start queue worker
ignite queue:work

# Process specific queue
ignite queue:work --queue=payments

# Process multiple queues
ignite queue:work --queue=high,default,low

Defining Events

package events

type UserRegistered struct {

Creating Listeners

package listeners

type SendWelcomeEmail struct{}

func (l *SendWelcomeEmail) Handle(event *events.UserRegistered) {
    // Send welcome email to event.User
}

// Queue listener
func (l *SendWelcomeEmail) ShouldQueue() bool {
    return true
}

Registering Listeners

import "github.com/sazzadh88/ignite/events"

events.Listen(&UserRegistered{}, &SendWelcomeEmail{})
events.Listen(&UserRegistered{}, &LogRegistration{})

Dispatching Events

events.Dispatch(&UserRegistered{User: user})

Event Subscribers

type UserEventSubscriber struct{}

func (s *UserEventSubscriber) Subscribe(dispatcher *events.Dispatcher) {
    dispatcher.Listen(&UserRegistered{}, s.OnUserRegistered)
    dispatcher.Listen(&UserDeleted{}, s.OnUserDeleted)

func (s *UserEventSubscriber) OnUserRegistered(e *UserRegistered) {
    // Handle registration
}

func (s *UserEventSubscriber) OnUserDeleted(e *UserDeleted) {
    // Handle deletion
}

Mail

Send beautiful emails with multiple transports (SMTP, SES, Mailgun):

Creating Mailables

package mail

import "github.com/sazzadh88/ignite/mail"

type WelcomeEmail struct {
    User *User
}

func (m *WelcomeEmail) Build() *mail.Message {
    return mail.
        To(m.User.Email).
        Subject("Welcome to Ignite!").
        View("emails.welcome", map[string]interface{}{
            "user": m.User,
        })
}

Sending Mail

import "github.com/sazzadh88/ignite/mail"

// Send mailable
mail.Send(&WelcomeEmail{User: user})

// Queue mailable
mail.Queue(&WelcomeEmail{User: user})

// Send later
mail.Later(60, &WelcomeEmail{User: user}) // 60 seconds

// Simple email
mail.
    To("user@example.com").
    Subject("Hello").
    Text("Email body").
    Send()

Attachments

func (m *InvoiceEmail) Build() *mail.Message {
    return mail.
        To(m.User.Email).
        Subject("Your Invoice").
        View("emails.invoice").
        Attach("/path/to/invoice.pdf")
}

Notifications

Send notifications across multiple channels (mail, SMS, Slack, database):

Creating Notifications

package notifications

type InvoicePaid struct {
    Invoice *Invoice
}

func (n *InvoicePaid) Via(notifiable interface{}) []string {
    return []string{"mail", "database"}
}

func (n *InvoicePaid) ToMail(notifiable interface{}) *mail.Message {
    return mail.
        Subject("Invoice Paid").
        View("emails.invoice-paid", map[string]interface{}{
            "invoice": n.Invoice,
        })
}

func (n *InvoicePaid) ToDatabase(notifiable interface{}) map[string]interface{} {
    return map[string]interface{}{
        "invoice_id": n.Invoice.ID,
        "amount":     n.Invoice.Amount,
    }
}

Sending Notifications

import "github.com/sazzadh88/ignite/notifications"

// Send to user
user.Notify(&InvoicePaid{Invoice: invoice})

// Send to multiple users
notifications.Send(users, &InvoicePaid{Invoice: invoice})

Task Scheduling

Schedule recurring tasks with a fluent cron-style scheduler:

Defining Schedules

package main

import "github.com/sazzadh88/ignite/schedule"

func RegisterSchedule(s *schedule.Schedule) {
    // Run every minute
    s.Call(cleanupTempFiles).EveryMinute()

    // Run hourly
    s.Call(sendReports).Hourly()

    // Run daily at specific time
    s.Call(backupDatabase).Daily().At("02:00")

    // Run on specific days
    s.Call(weeklyReport).Weekly().Monday().At("08:00")

    // Custom cron expression
    s.Call(complexTask).Cron("0 */6 * * *")

    // Prevent overlaps
    s.Call(longTask).EveryFiveMinutes().WithoutOverlapping()

    // Run in background
    s.Call(task).Hourly().RunInBackground()
}

Frequency Options

.EveryMinute()
.EveryFiveMinutes()
.EveryTenMinutes()
.EveryThirtyMinutes()
.Hourly()
.Daily()
.Weekly()
.Monthly()
.Yearly()
.Weekdays()
.Weekends()
.Monday(), .Tuesday(), etc.

Running the Scheduler

ignite schedule:run

HTTP Client

Fluent HTTP client with retries, middleware, and macros:

Making Requests

import "github.com/sazzadh88/ignite/http/client"

// GET request
response := client.Get("https://api.example.com/users")

// POST request
response := client.Post("https://api.example.com/users", map[string]interface{}{
    "name":  "John",
    "email": "john@example.com",
})

// With headers
response := client.
    WithHeaders(map[string]string{
        "Authorization": "Bearer " + token,
    }).
    Get("https://api.example.com/users")

// JSON response
var users []User
response.JSON(&users)

Retry and Timeout

response := client.
    Retry(3, 100). // 3 retries, 100ms delay
    Timeout(30).      // 30 seconds
    Get(url)

Rate Limiting

Flexible rate limiting for APIs and routes:

import "github.com/sazzadh88/ignite/ratelimit"

// Route rate limit
router.
    Get("/api/users", handler).
    Middleware(ratelimit.Throttle(60, 1)) // 60 per minute

// Per-user rate limit
limiter := ratelimit.For("api").
    By(func(req *http.Request) string {
        return req.User().ID
    }).
    Limit(100).
    PerMinute()

// Manual rate limiting
if limiter.TooManyAttempts(key) {
    return req.JSON(map[string]string{
        "error": "Too many requests",
    }).Status(429)
}

limiter.Hit(key)

File Storage

Unified file storage interface for local and cloud storage (S3):

import "github.com/sazzadh88/ignite/storage"

// Store file
path := storage.Put("avatars/user.jpg", fileData)

// Get file
data := storage.Get("avatars/user.jpg")

// Delete file
storage.Delete("avatars/user.jpg")

// Check existence
if storage.Exists("avatars/user.jpg") {
    // File exists
}

// Get URL
url := storage.URL("avatars/user.jpg")

// Use specific disk
storage.Disk("s3").Put("file.jpg", data)

Testing

Built-in testing utilities for HTTP, database, and more:

import "github.com/sazzadh88/ignite/testing"

func TestUserCreation(t *testing.T) {
    // HTTP testing
    response := testing.
        Post("/users", map[string]interface{}{
            "name":  "John",
            "email": "john@example.com",
        })

    response.
        AssertStatus(201).
        AssertJSON(map[string]interface{}{
            "name": "John",
        })

    // Database assertions
    testing.AssertDatabaseHas("users", map[string]interface{}{
        "email": "john@example.com",
    })
}
