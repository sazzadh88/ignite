Authentication

Ignite includes a complete authentication system out of the box with support for session-based and token-based authentication, plus a powerful authorization system with Gates and Policies.

Configuration

Authentication settings are configured in config/auth.go:

package config

import "github.com/sazzadh88/ignite/auth"

func Auth() *auth.Config {
    return &auth.Config{
        Defaults: auth.Defaults{
            Guard: "web",
        },
        Guards: map[string]auth.Guard{
            "web": {
                Driver:   "session",
                Provider: "users",
            },
            "api": {
                Driver:   "token",
                Provider: "users",
            },
        },
        Providers: map[string]auth.Provider{
            "users": {
                Driver: "eloquent",
                Model:  "models.User",
            },
        },
    }
}

Guards

Guards define how users are authenticated for each request. Ignite includes session and token guards:

Session Guard (Web)

Session-based authentication stores the user ID in the session:

import "github.com/sazzadh88/ignite/auth"

// Attempt to authenticate
if auth.Attempt(map[string]interface{}{
    "email":    req.Input("email"),
    "password": req.Input("password"),
}) {
    // Authentication passed
    return req.Redirect("/dashboard")
}

// Authentication failed
return req.Back().WithErrors(map[string]string{
    "email": "Invalid credentials",
})

Token Guard (API)

Token-based authentication uses bearer tokens for API requests:

// Generate token after login
user := /* get user */
token := auth.Guard("api").CreateToken(user)

return req.JSON(map[string]interface{}{
    "access_token": token,
    "token_type":   "Bearer",
})

// Client includes token in requests:
// Authorization: Bearer TOKEN_HERE

Using Different Guards

// Use default guard
user := auth.User()

// Use specific guard
user := auth.Guard("api").User()

// Check if authenticated
if auth.Check() {
    // User is authenticated
}

Authentication

Login

package controllers

import (
    "github.com/sazzadh88/ignite/auth"
    "github.com/sazzadh88/ignite/http"
)

func Login(req *http.Request) *http.Response {
    credentials := map[string]interface{}{
        "email":    req.Input("email"),
        "password": req.Input("password"),
    }

    // Simple login
    if auth.Attempt(credentials) {
        return req.Redirect("/dashboard")
    }

    // Login with "remember me"
    if auth.Attempt(credentials, req.Has("remember")) {
        return req.Redirect("/dashboard")
    }

    return req.Back().WithErrors(map[string]string{
        "email": "Invalid credentials",
    })
}

Logout

func Logout(req *http.Request) *http.Response {
    auth.Logout()
    return req.Redirect("/")
}

Manual Login

// Login a user by ID
auth.LoginUsingID(123)

// Login a user instance
auth.Login(user)

// Login for a single request (stateless)
auth.Once(credentials)

Retrieving the Authenticated User

// Get authenticated user
user := auth.User()

// Get user ID
id := auth.ID()

// Check authentication status
if auth.Check() {
    // User is logged in
}

if auth.Guest() {
    // User is not logged in
}

Password Hashing

Ignite uses bcrypt for secure password hashing:

import "github.com/sazzadh88/ignite/hashing"

// Hash a password
hashed := hashing.Make("secret-password")

// Verify a password
if hashing.Check("secret-password", hashed) {
    // Password matches
}

// Check if rehashing is needed
if hashing.NeedsRehash(hashed) {
    // Rehash the password
    newHashed := hashing.Make(password)
}

User Registration Example

func Register(req *http.Request) *http.Response {
    // Validate request...

    user := User{
        Name:     req.Input("name"),
        Email:    req.Input("email"),
        Password: hashing.Make(req.Input("password")),
    }

    database.Create(&user)

    // Log the user in
    auth.Login(user)

    return req.Redirect("/dashboard")
}

Protecting Routes

Use the auth middleware to require authentication:

import "github.com/sazzadh88/ignite/middleware"

// Protect a single route
router.
    Get("/dashboard", dashboardHandler).
    Middleware(middleware.Auth())

// Protect a route group
router.Group(func(r *routing.Router) {
    r.Middleware(middleware.Auth())

    r.Get("/dashboard", dashboardHandler)
    r.Get("/profile", profileHandler)
    r.Post("/settings", settingsHandler)
})

// Specify guard
router.
    Get("/api/users", apiUsersHandler).
    Middleware(middleware.Auth("api"))

Guest Middleware

Redirect authenticated users away from login/register pages:

router.
    Get("/login", loginHandler).
    Middleware(middleware.Guest())

API Tokens

Issue and manage personal access tokens for API authentication:

Creating Tokens

import "github.com/sazzadh88/ignite/auth"

// Create a token
token := user.CreateToken("token-name")

// Create token with abilities (scopes)
token := user.CreateToken("token-name", []string{
    "posts:read",
    "posts:write",
})

Token Abilities

// Check if user can perform ability
if user.TokenCan("posts:write") {
    // Token has ability
}

Revoking Tokens

// Revoke specific token
user.RevokeToken(tokenID)

// Revoke all tokens
user.RevokeAllTokens()

Authorization - Gates

Gates provide a simple closure-based approach to authorization:

Defining Gates

import "github.com/sazzadh88/ignite/gate"

// In a service provider
gate.Define("update-post", func(user *User, post *Post) bool {
    return user.ID == post.UserID
})

gate.Define("view-admin", func(user *User) bool {
    return user.IsAdmin
})

Checking Abilities

// Check if user is authorized
if gate.Allows("update-post", post) {
    // User can update post
}

if gate.Denies("update-post", post) {
    // User cannot update post
}

// Authorize or throw
gate.Authorize("update-post", post) // Throws 403 if denied

Before and After Hooks

// Run before all gate checks
gate.Before(func(user *User, ability string) *bool {
    if user.IsSuperAdmin {
        allow := true
        return &allow // Allow everything
    }
    return nil // Continue to gate check
})

// Run after all gate checks
gate.After(func(user *User, ability string, result bool) bool {
    // Log authorization attempt
    return result
})

Authorization - Policies

Policies organize authorization logic around a particular model:

Creating Policies

./ignite make:policy PostPolicy

This creates app/policies/PostPolicy.go:

package policies

import "app/models"

type PostPolicy struct{}

func (p *PostPolicy) View(user *models.User, post *models.Post) bool {
    return true // Anyone can view
}

func (p *PostPolicy) Update(user *models.User, post *models.Post) bool {
    return user.ID == post.UserID
}

func (p *PostPolicy) Delete(user *models.User, post *models.Post) bool {
    return user.ID == post.UserID || user.IsAdmin
}

func (p *PostPolicy) Create(user *models.User) bool {
    return user.Active // Only active users can create
}

Registering Policies

import "github.com/sazzadh88/ignite/gate"

gate.Policy(&Post{}, &PostPolicy{})

Using Policies

// Check via gate
if gate.Allows("update", post) {
    // User can update
}

// Check via user model
if user.Can("update", post) {
    // User can update
}

// Authorize or throw
user.Authorize("update", post)

Policy Middleware

router.
    Put("/posts/{id}", updatePostHandler).
    Middleware(middleware.Can("update", "post"))

Email Verification

Require users to verify their email addresses:

// User model
type User struct {
    orm.Model
    EmailVerifiedAt *time.Time
}

// Implement email verifiable
func (u *User) HasVerifiedEmail() bool {
    return u.EmailVerifiedAt != nil
}

func (u *User) MarkEmailAsVerified() {
    now := time.Now()
    u.EmailVerifiedAt = &now
    database.Save(u)
}

// Protect routes
router.
    Get("/dashboard", dashboardHandler).
    Middleware(middleware.Verified())
